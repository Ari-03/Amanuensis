cmake_minimum_required(VERSION 3.20)
project(AmanuensisS1Mini LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "Pass -DLLAMA_SOURCE=/path/to/pinned/llama.cpp")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(S1MiniHelper main.cpp)
target_include_directories(S1MiniHelper PRIVATE "${LLAMA_SOURCE}/vendor")
target_link_libraries(S1MiniHelper PRIVATE llama)
target_compile_options(S1MiniHelper PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
